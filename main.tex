% REFERÊNCIAS
% Alinhamento à esquerda na página e espaçamento reduzido no sumário
\makeatletter
\renewcommand{\bibsection}{%
    \begingroup
    % Força alinhamento à esquerda (não centralizado)
    \def\printchaptertitle##1{\chaptitlefont\ABNTEXchapterupperifneeded{##1}}
    \chapter*{\bibname}
    \bibmark
    \ifnobibintoc\else
    \phantomsection
    % Remove pulo de linha extra para o sumário (assemelhando-se às seções)
    \addtocontents{toc}{\protect\vspace{-\cftbeforechapterskip}}
    \addcontentsline{toc}{chapter}{\texorpdfstring{\MakeTextUppercase{\bibname}}{\bibname}}
    \fi
    \prebibhook
    \endgroup
}
\makeatother
